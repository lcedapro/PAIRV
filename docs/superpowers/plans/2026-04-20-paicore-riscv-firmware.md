# PaiCore RISC-V Firmware Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the RISC-V firmware for PaiCore neuromorphic chip: Protobuf host communication, FreeRTOS multi-thread inference, NOC frame construction, and host-side Python demo.

**Architecture:** FreeRTOS tasks (UART RX/TX, dispatch, 2 workers) communicate via queues and semaphores. Host↔RISC-V uses EmbeddedProto over UART with length-prefixed framing. Workers translate [T,C,H,W] tensors to chip-native [T,CoreAddr,AxonAddr] work frames via FLASH lookup table.

**Tech Stack:** nuclei-sdk (RISC-V GCC), EmbeddedProto (C++11 protobuf), FreeRTOS, Python (pyserial + protobuf)

**Spec:** `docs/superpowers/specs/2026-04-20-riscv-embedded-software-design.md`

> **路径说明：** 应用目录为 `application/freertos/embedded_proto/`，EmbeddedProto 位于 `third_party/EmbeddedProto/`，`NUCLEI_SDK_ROOT = ../../..`，`CORE = n307fd`。

```protobuf
message InitRequest {
  uint32 thread_id = 1;
}

message InferRequest {
  uint32         thread_id = 1;
  uint32         timesteps = 2;
  bytes          data      = 3;   // [T,C,H,W] flattened uint8 tensor
}

message HostMessage {
  oneof payload {
    InitRequest  init_req  = 1;
    InferRequest infer_req = 2;
  }
}

message InitResponse {
  uint32 thread_id = 1;
  bool   success   = 2;
}

message InferResponse {
  uint32 thread_id = 1;
  bool   success   = 2;
  bytes  data      = 3;   // [T,C,H,W] flattened uint8 output tensor

}

message DeviceMessage {
  oneof payload {
    InitResponse  init_resp  = 1;
    InferResponse infer_resp = 2;
  }
}
```

Note: `repeated uint8` is not valid protobuf. Use `bytes` instead — it is the canonical way to carry raw byte arrays in protobuf3, and EmbeddedProto supports it via `FieldStringBytes` with fixed-size backing.

```c
#ifndef RINGBUF_H
#define RINGBUF_H

#include <stdint.h>
#include <stddef.h>

typedef struct {
    uint64_t *buf;
    volatile uint32_t head;  // written by producer (ISR)
    volatile uint32_t tail;  // read by consumer (task)
    uint32_t capacity;       // must be power of 2
} ringbuf_t;

void     ringbuf_init(ringbuf_t *rb, uint64_t *storage, uint32_t capacity);
int      ringbuf_push(ringbuf_t *rb, uint64_t val);   // returns 0 on success, -1 if full
int      ringbuf_pop(ringbuf_t *rb, uint64_t *val);    // returns 0 on success, -1 if empty
uint32_t ringbuf_count(const ringbuf_t *rb);
void     ringbuf_clear(ringbuf_t *rb);

#endif
```

```c
// application/freertos/embedded_proto/src/ringbuf.c
#include "ringbuf.h"

void ringbuf_init(ringbuf_t *rb, uint64_t *storage, uint32_t capacity)
{
    rb->buf = storage;
    rb->head = 0;
    rb->tail = 0;
    rb->capacity = capacity;
}

int ringbuf_push(ringbuf_t *rb, uint64_t val)
{
    uint32_t next = (rb->head + 1) & (rb->capacity - 1);
    if (next == rb->tail) return -1;
    rb->buf[rb->head] = val;
    rb->head = next;
    return 0;
}

int ringbuf_pop(ringbuf_t *rb, uint64_t *val)
{
    if (rb->head == rb->tail) return -1;
    *val = rb->buf[rb->tail];
    rb->tail = (rb->tail + 1) & (rb->capacity - 1);
    return 0;
}

uint32_t ringbuf_count(const ringbuf_t *rb)
{
    return (rb->head - rb->tail) & (rb->capacity - 1);
}

void ringbuf_clear(ringbuf_t *rb)
{
    rb->tail = rb->head;
}
```

```c
    assert(ringbuf_count(&rb) == CAP - 1);
    assert(ringbuf_push(&rb, 99) == -1); // full

        assert(val == (uint64_t)i);

    ringbuf_t rb;
```

```c
{
    *irq_reg = 0;
    *irq_en  = 1;
}

void noc_write_frame(uint32_t high, uint32_t low)
{
    *fifo_down = high;
    *fifo_down = low;
}

uint64_t noc_read_frame(void)
{
    uint32_t high = (uint32_t)(*fifo_up);
    uint32_t low  = (uint32_t)(*fifo_up);
    return ((uint64_t)high << 32) | low;
}

{
    return (*irq_reg) != 0;
}

void noc_irq_enable(void)  { *irq_en = 1; }
void noc_irq_disable(void) { *irq_en = 0; }
```

```c
// application/freertos/embedded_proto/src/frame_builder.c
#include "frame_builder.h"

frame64_t frame_build_work(
    uint16_t timestep,
    uint8_t  core_xy,
    uint8_t  core_x,
    uint8_t  core_y,
    uint8_t  copy_xy,
    uint8_t  copy_x,
    uint8_t  copy_y,
    uint16_t axon_addr,
    uint8_t  data)
{
    frame64_t f;
    // 64-bit layout (high=bits[63:32], low=bits[31:0]):
    // [63:62]=0b10  [61]=0(type1)  [60]=TS[7]
    // [59:54]=Core_XY(6)  [53:48]=Core_X(6)  [47:42]=Core_Y(6)
    // [41:36]=Copy_XY(6)  [35:32]=Copy_X[5:2]
    // [31:30]=Copy_X[1:0]  [29:24]=Copy_Y(6)
    // LOAD[23:0]: [23:17]=TS[6:0]  [16:8]=Axon(9)  [7:0]=Data
    f.high  = (0x2u << 30);                          // [63:62]=0b10, [61]=0
    f.high |= ((uint32_t)((timestep >> 7) & 0x1) << 28); // [60]=TS[7]
    f.high |= ((uint32_t)(core_xy  & 0x3F) << 22);  // [59:54]
    f.high |= ((uint32_t)(core_x   & 0x3F) << 16);  // [53:48]
    f.high |= ((uint32_t)(core_y   & 0x3F) << 10);  // [47:42]
    f.high |= ((uint32_t)(copy_xy  & 0x3F) << 4);   // [41:36]
    f.high |= ((uint32_t)(copy_x   >> 2)   & 0xF);  // [35:32]=Copy_X[5:2]

    f.low   = ((uint32_t)(copy_x   & 0x3)  << 30);  // [31:30]=Copy_X[1:0]
    f.low  |= ((uint32_t)(copy_y   & 0x3F) << 24);  // [29:24]
    f.low  |= ((uint32_t)(timestep & 0x7F) << 17);  // [23:17]=TS[6:0]
    f.low  |= ((uint32_t)(axon_addr & 0x1FF) << 8); // [16:8]=Axon(9-bit)
    f.low  |= ((uint32_t)(data     & 0xFF));         // [7:0]

    return f;
}

frame64_t frame_build_control(
    uint8_t  ctrl_type,
    uint8_t  core_xy,
    uint8_t  core_x,
    uint8_t  core_y,
    uint8_t  copy_xy,
    uint8_t  copy_x,
    uint8_t  copy_y,
    uint32_t load)
{
    frame64_t f;
    // 64-bit layout:
    // [63:62]=0b11  [61:60]=type(2)
    // [59:54]=Core_XY(6)  [53:48]=Core_X(6)  [47:42]=Core_Y(6)
    // [41:36]=Copy_XY(6)  [35:32]=Copy_X[5:2]
    // [31:30]=Copy_X[1:0]  [29:24]=Copy_Y(6)
    // LOAD[23:0]
    f.high  = (0x3u << 30);                          // [63:62]=0b11
    f.high |= ((uint32_t)(ctrl_type & 0x3)  << 28); // [61:60]
    f.high |= ((uint32_t)(core_xy   & 0x3F) << 22); // [59:54]
    f.high |= ((uint32_t)(core_x    & 0x3F) << 16); // [53:48]
    f.high |= ((uint32_t)(core_y    & 0x3F) << 10); // [47:42]
    f.high |= ((uint32_t)(copy_xy   & 0x3F) << 4);  // [41:36]
    f.high |= ((uint32_t)(copy_x    >> 2)   & 0xF); // [35:32]=Copy_X[5:2]

    f.low   = ((uint32_t)(copy_x    & 0x3)  << 30); // [31:30]=Copy_X[1:0]
    f.low  |= ((uint32_t)(copy_y    & 0x3F) << 24); // [29:24]
    f.low  |= (load & 0x00FFFFFF);                   // [23:0]

    return f;
}

uint8_t frame_get_header(uint32_t high)
{
    return (high >> 30) & 0x3;
}

uint8_t frame_get_ctrl_type(uint32_t high)
{
    return (high >> 28) & 0x3;
}

uint32_t frame_get_ctrl_load(uint32_t high, uint32_t low)
{
    (void)high;
    return low & 0x00FFFFFF;
}
```

```c
// application/freertos/embedded_proto/src/uart_frame.c
#include "uart_frame.h"
#include "paicore_config.h"

void uart_frame_parser_init(uart_frame_parser_t *p, uint8_t *buf, uint16_t buf_size)
{
    p->state = UF_WAIT_MAGIC_0;
    p->payload_len = 0;
    p->payload_idx = 0;
    p->payload_buf = buf;
    p->payload_buf_size = buf_size;
}

void uart_frame_parser_reset(uart_frame_parser_t *p)
{
    p->state = UF_WAIT_MAGIC_0;
    p->payload_len = 0;
    p->payload_idx = 0;
}

int uart_frame_parser_feed(uart_frame_parser_t *p, uint8_t byte)
{
    switch (p->state) {
    case UF_WAIT_MAGIC_0:
        if (byte == UART_FRAME_MAGIC_B0) p->state = UF_WAIT_MAGIC_1;
        break;
    case UF_WAIT_MAGIC_1:
        p->state = (byte == UART_FRAME_MAGIC_B1) ? UF_WAIT_MAGIC_2 : UF_WAIT_MAGIC_0;
        break;
    case UF_WAIT_MAGIC_2:
        p->state = (byte == UART_FRAME_MAGIC_B2) ? UF_WAIT_MAGIC_3 : UF_WAIT_MAGIC_0;
        break;
    case UF_WAIT_MAGIC_3:
        p->state = (byte == UART_FRAME_MAGIC_B3) ? UF_WAIT_LEN_LO : UF_WAIT_MAGIC_0;
        break;
    case UF_WAIT_LEN_LO:
        p->payload_len = byte;
        p->state = UF_WAIT_LEN_HI;
        break;
    case UF_WAIT_LEN_HI:
        p->payload_len |= ((uint16_t)byte << 8);
        p->payload_idx = 0;
        if (p->payload_len == 0 || p->payload_len > p->payload_buf_size) {
            p->state = UF_WAIT_MAGIC_0;
        } else {
            p->state = UF_RECV_PAYLOAD;
        }
        break;
    case UF_RECV_PAYLOAD:
        p->payload_buf[p->payload_idx++] = byte;
        if (p->payload_idx >= p->payload_len) {
            p->state = UF_WAIT_MAGIC_0;
            return 1;
        }
        break;
    }
    return 0;
}

const uint8_t *uart_frame_parser_payload(const uart_frame_parser_t *p)
{
    return p->payload_buf;
}

uint16_t uart_frame_parser_payload_len(const uart_frame_parser_t *p)
{
    return p->payload_len;
}

int uart_frame_build(const uint8_t *payload, uint16_t payload_len,
                     uint8_t *out_buf, uint16_t out_buf_size)
{
    uint16_t total = 4 + 2 + payload_len;
    if (total > out_buf_size) return -1;

    out_buf[0] = UART_FRAME_MAGIC_B0;
    out_buf[1] = UART_FRAME_MAGIC_B1;
    out_buf[2] = UART_FRAME_MAGIC_B2;
    out_buf[3] = UART_FRAME_MAGIC_B3;
    out_buf[4] = (uint8_t)(payload_len & 0xFF);
    out_buf[5] = (uint8_t)(payload_len >> 8);

    for (uint16_t i = 0; i < payload_len; i++) {
        out_buf[6 + i] = payload[i];
    }
    return (int)total;
}
```

  ```c
          uint64_t frame = noc_read_frame();
          uint32_t high  = (uint32_t)(frame >> 32);
          uint32_t low   = (uint32_t)(frame);

          uint8_t hdr = frame_get_header(high);

          if (hdr == 0x3) {  // control frame
              uint8_t ctype = frame_get_ctrl_type(high);
              if (ctype == 0x02) {  // complete frame, LOAD[23:0] = thread_id
                  uint32_t tid = frame_get_ctrl_load(high, low);
                  if (tid < MAX_THREADS) {
                      xSemaphoreGiveFromISR(done_sem[tid], &higher_woken);
                  }
              }
          } else if (hdr == 0x2) {  // work frame (output data)
              // CoreAddr fields are zeroed by NOC; use active_thread instead
              ringbuf_push(&output_ringbuf[active_thread], frame);
          }
      }

      noc_irq_enable();

      portYIELD_FROM_ISR(higher_woken);

      RESTORE_IRQ_CSR_CONTEXT();
  }
  ```

  ```c
  // application/freertos/embedded_proto/src/worker_task.c
  #include <stdio.h>
  #include "FreeRTOS.h"
  #include "task.h"
  #include "semphr.h"
  #include "worker_task.h"
  #include "noc.h"
  #include "frame_builder.h"
  #include "neuron_map.h"
  #include "paicore_config.h"

  extern const input_neuron_mapping_t  input_neuron_map_table[];
  // One output map per thread, indexed by axon_addr
  extern const output_neuron_mapping_t output_neuron_map_table[MAX_THREADS][MAX_AXON_ADDR];
  static neuron_map_ctx_t g_map_ctx[MAX_THREADS];
  static int g_map_inited = 0;

  extern volatile uint32_t active_thread;  // defined in snn_isr.c

  static void ensure_map_init(void)
  {
      if (!g_map_inited) {
          for (uint32_t t = 0; t < MAX_THREADS; t++) {
              neuron_map_init(&g_map_ctx[t],
                              input_neuron_map_table,
                              output_neuron_map_table[t],
                              MAX_C, MAX_H, MAX_W, MAX_AXON_ADDR);
          }
          g_map_inited = 1;
      }
  }

  static void send_init_frames(uint32_t thread_id)
  {
      // Broadcast init control frame to all cores for this thread
      // Core address 0,0,0 with copy broadcast — adjust per network topology
      frame64_t f = frame_build_control(0x01, 0, 0, 0, 0, 0, 0, thread_id);
      noc_write_frame(f.high, f.low);
  }

  static void send_work_frames(const worker_msg_t *req)
  {
      ensure_map_init();
      uint32_t T = req->timesteps;
      uint32_t CHW = MAX_C * MAX_H * MAX_W;

      for (uint32_t t = 0; t < T; t++) {
          for (uint32_t chw = 0; chw < CHW && chw < req->data_len / T; chw++) {
              uint8_t val = req->data[t * (req->data_len / T) + chw];
              if (val == 0) continue;  // skip zero spikes

              uint32_t c = chw / (MAX_H * MAX_W);
              uint32_t h = (chw / MAX_W) % MAX_H;
              uint32_t w = chw % MAX_W;

              const input_neuron_mapping_t *m = neuron_map_lookup(&g_map_ctx[req->thread_id], c, h, w);
              frame64_t f = frame_build_work(
                  (uint16_t)t,
                  m->core_xy, m->core_x, m->core_y,
                  0, 0, 0,  // copy fields — set per network
                  m->axon_addr, val
              );
              noc_write_frame(f.high, f.low);
          }
      }

      // Set active_thread before sync frame so ISR routes output frames correctly
      active_thread = req->thread_id;
      // Send sync frame
      frame64_t sync = frame_build_control(0x00, 0, 0, 0, 0, 0, 0, T);
      noc_write_frame(sync.high, sync.low);
  }

  static void collect_output(uint32_t thread_id, worker_resp_t *resp)
  {
      ensure_map_init();
      resp->data_len = 0;

      uint64_t frame;
      while (ringbuf_pop(&output_ringbuf[thread_id], &frame) == 0) {
          uint32_t low  = (uint32_t)(frame);

          // CoreAddr fields are zeroed by NOC; only axon_addr is meaningful
          uint16_t axon = (low >> 8) & 0x1FF;
          uint8_t  data = (low     ) & 0xFF;

          uint32_t c, h, w;
          if (output_map_lookup(&g_map_ctx[thread_id], axon, &c, &h, &w) == 0) {
              uint32_t idx = c * MAX_H * MAX_W + h * MAX_W + w;
              if (idx < MAX_NEURONS) {
                  resp->data[idx] = data;
                  if (idx + 1 > resp->data_len) resp->data_len = idx + 1;
              }
          }
      }
  }

  void worker_task(void *param)
  {
      uint32_t thread_id = (uint32_t)(uintptr_t)param;
      worker_msg_t  req;
      worker_resp_t resp;

      printf("[worker%lu] started\n", (unsigned long)thread_id);

      for (;;) {
          xQueueReceive(input_queue[thread_id], &req, portMAX_DELAY);

          if (req.type == MSG_INIT_REQ) {
              xSemaphoreTake(noc_mutex, portMAX_DELAY);
              send_init_frames(thread_id);
              xSemaphoreGive(noc_mutex);

              BaseType_t ok = xSemaphoreTake(done_sem[thread_id],
                                             pdMS_TO_TICKS(INIT_TIMEOUT_MS));
              resp.type      = MSG_INIT_REQ;
              resp.thread_id = thread_id;
              resp.success   = (ok == pdTRUE) ? 1 : 0;
              resp.data_len  = 0;
              resp.cycles    = 0;

          } else {  // MSG_INFER_REQ
              uint32_t t0 = (uint32_t)xTaskGetTickCount();

              xSemaphoreTake(noc_mutex, portMAX_DELAY);
              send_work_frames(&req);
              xSemaphoreGive(noc_mutex);

              BaseType_t ok = xSemaphoreTake(done_sem[thread_id],
                                             pdMS_TO_TICKS(INFER_TIMEOUT_MS));

              resp.type      = MSG_INFER_REQ;
              resp.thread_id = thread_id;
              resp.success   = (ok == pdTRUE) ? 1 : 0;
              resp.data_len  = 0;
              resp.cycles    = (uint32_t)xTaskGetTickCount() - t0;

              if (ok == pdTRUE) {
                  collect_output(thread_id, &resp);
              }
          }

          xQueueSend(tx_queue, &resp, portMAX_DELAY);
      }
  }
  ```

  ```c
  // UART ISR → task byte queue
  static QueueHandle_t uart_byte_queue;

  void UART0_IRQHandler(void)
  {
      SAVE_IRQ_CSR_CONTEXT();
      uint32_t status = UART0->IP;
      if (status & UART_IP_RXWM) {
          uint32_t rxfifo_val;
          do {
              rxfifo_val = UART0->RXFIFO;
              if (!(rxfifo_val & UART_RXFIFO_EMPTY)) {
                  uint8_t b = (uint8_t)(rxfifo_val & 0xFF);
                  BaseType_t woken = pdFALSE;
                  xQueueSendFromISR(uart_byte_queue, &b, &woken);
                  portYIELD_FROM_ISR(woken);
              }
          } while (!(rxfifo_val & UART_RXFIFO_EMPTY));
      }
      RESTORE_IRQ_CSR_CONTEXT();
  }

  void uart_rx_task(void *param)
  {
      (void)param;

      uart_byte_queue = xQueueCreate(UART_RX_BUF_SIZE, sizeof(uint8_t));

      static uint8_t frame_buf[PROTO_BUF_SIZE];
      uart_frame_parser_t parser;
      uart_frame_parser_init(&parser, frame_buf, sizeof(frame_buf));

      for (;;) {
          uint8_t byte;
          xQueueReceive(uart_byte_queue, &byte, portMAX_DELAY);

          if (!uart_frame_parser_feed(&parser, byte)) continue;

          // Complete frame received — deserialize
          const uint8_t *payload = uart_frame_parser_payload(&parser);
          uint16_t       plen    = uart_frame_parser_payload_len(&parser);

          EmbeddedProto::ReadBufferFixedSize<PROTO_BUF_SIZE> rbuf;
          memcpy(rbuf.get_data(), payload, plen);
          rbuf.set_bytes_written(plen);

          paicore::HostMessage msg;
          if (msg.deserialize(rbuf) != EmbeddedProto::Error::NO_ERRORS) continue;

          worker_msg_t wmsg = {0};

          if (msg.has_init_req()) {
              wmsg.type      = MSG_INIT_REQ;
              wmsg.thread_id = msg.get_init_req().get_thread_id();
          } else if (msg.has_infer_req()) {
              wmsg.type      = MSG_INFER_REQ;
              wmsg.thread_id = msg.get_infer_req().get_thread_id();
              wmsg.timesteps = msg.get_infer_req().get_timesteps();
              auto &d = msg.get_infer_req().get_data();
              wmsg.data_len  = d.get_length();
              memcpy(wmsg.data, d.get_data(), wmsg.data_len);
          } else {
              continue;
          }

          xQueueSend(dispatch_queue, &wmsg, portMAX_DELAY);
      }
  }
  ```

  ```c
      (void)param;

      static uint8_t frame_buf[PROTO_BUF_SIZE + 6];

      for (;;) {
          xQueueReceive(tx_queue, &resp, portMAX_DELAY);

          EmbeddedProto::WriteBufferFixedSize<PROTO_BUF_SIZE> wbuf;
          paicore::DeviceMessage dmsg;

          if (resp.type == MSG_INIT_REQ) {
              auto &ir = dmsg.mutable_init_resp();
              ir.set_thread_id(resp.thread_id);
              ir.set_success(resp.success);
          } else {
              auto &ir = dmsg.mutable_infer_resp();
              ir.set_thread_id(resp.thread_id);
              ir.set_success(resp.success);
              ir.set_cycles(resp.cycles);
              auto &d = ir.mutable_data();
              d.set(resp.data, resp.data_len);
          }

          if (dmsg.serialize(wbuf) != EmbeddedProto::Error::NO_ERRORS) continue;

          uint16_t plen = wbuf.get_size();
          memcpy(proto_buf, wbuf.get_data(), plen);

          int flen = uart_frame_build(proto_buf, plen, frame_buf, sizeof(frame_buf));
          if (flen > 0) {
              uart_write_bytes(frame_buf, (uint16_t)flen);
  ```

  ```python
  import serial
  import struct
  from paicore_pb2 import HostMessage, DeviceMessage

  MAGIC = b'PAIC'
  BAUDRATE = 3_000_000


  class PaiCoreClient:
      def __init__(self, port: str):
          self.ser = serial.Serial(port, BAUDRATE, timeout=5)

      def close(self):
          self.ser.close()

      def _send(self, msg: HostMessage):
          payload = msg.SerializeToString()
          header = MAGIC + struct.pack('<H', len(payload))
          self.ser.write(header + payload)

      def _recv(self) -> DeviceMessage:
          buf = b''
          while True:
              b = self.ser.read(1)
              if not b:
                  raise TimeoutError("No response from device")
              buf += b
              if len(buf) >= 4 and buf[-4:] == MAGIC:
                  break
          raw_len = self.ser.read(2)
          length = struct.unpack('<H', raw_len)[0]
          payload = self.ser.read(length)
          resp = DeviceMessage()
          resp.ParseFromString(payload)
          return resp

      def send_init(self, thread_id: int) -> bool:
          msg = HostMessage()
          msg.init_req.thread_id = thread_id
          self._send(msg)
          resp = self._recv()
          return resp.init_resp.success

      def send_infer(self, thread_id: int, timesteps: int,
                     data: bytes) -> tuple:
          msg = HostMessage()
          msg.infer_req.thread_id = thread_id
          msg.infer_req.timesteps = timesteps
          msg.infer_req.data = data
          self._send(msg)
          resp = self._recv()
          return resp.infer_resp.success, resp.infer_resp.data, resp.infer_resp.cycles


  if __name__ == '__main__':
      import sys
  ```

  ```python
      length = struct.unpack('<H', data[4:6])[0]

      resp = DeviceMessage()
      resp.ParseFromString(payload)
      return resp
  ```

  ### Notes for Implementer

  1. **`repeated uint8` → `bytes`**: The spec originally used `repeated uint8` which is
   invalid protobuf3. This plan uses `bytes data = 3` instead — it is the correct way
  to carry raw byte arrays and EmbeddedProto supports it via `FieldStringBytes`.

  2. **Frame bit layout**: `frame_builder.c` implements the 64-bit frame layout from the
  spec. Copy_X spans the high/low 32-bit boundary ([35:32] in high, [31:30] in low) —
  this is handled correctly in the implementation. Verify all bit positions against the
  actual chip RTL before hardware testing.

  3. **`active_thread` in snn_isr.c**: The current design assumes only one thread is
  active at a time (enforced by noc_mutex). If multi-thread concurrent inference is
  needed in the future, `active_thread` must be replaced with a proper per-frame
  routing mechanism.

  4. **EmbeddedProto generated API**: The `uart_rx_task.c` and `uart_tx_task.c` use
  `has_init_req()`, `get_init_req()`, `mutable_init_resp()` etc. — these are the
  standard EmbeddedProto generated method names. Verify against the actual generated
  `paicore.h` after running protoc-gen-eams.

  5. **accel_mutex**: BatchNorm/Softmax accelerator peripheral calls are stubbed in
  `worker_task.c`. Implement `accel.c` / `accel.h` when peripheral register map is
  available.
